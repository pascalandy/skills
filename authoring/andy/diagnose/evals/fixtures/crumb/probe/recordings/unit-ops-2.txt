# /etc/systemd/system/preorder-export.timer
[Unit]
Description=Nightly preorder sheet for the kitchen

[Timer]
OnCalendar=*-*-* 21:00:00 America/Toronto
Persistent=true

[Install]
WantedBy=timers.target

# /etc/systemd/system/preorder-export.service
[Unit]
Description=Crumb preorder sheet for the kitchen
Wants=network-online.target
After=network-online.target

[Service]
Type=oneshot
User=crumb
Group=crumb
WorkingDirectory=/opt/crumb
ExecStart=/usr/bin/python3 -m crumb.export --for tomorrow
